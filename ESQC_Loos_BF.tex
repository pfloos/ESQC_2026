\begin{frame}{Expanding the molecular orbitals}
	\begin{block}{Linear combination of basis functions}
    	\begin{equation*}
    		\boxed{
    			\psi_p(\br)
    			=
    			\sum_{\mu=1}^{K}
    			C_{\mu p}\phi_\mu(\br)
    		}
    	\end{equation*}
    
    	\begin{itemize}
    		\item The basis functions $\{\phi_\mu\}$ are \alert{fixed}; the coefficients
    		      $C_{\mu p}$ are determined \alert{variationally}
    
    		\item Increasing $K$ improves the flexibility
    		      and, ideally, approaches the \alert{complete basis set limit (CBS)}
    
    		\item Basis sets contain the \alert{only empirical parameters} of an \textit{ab initio} 
    		      calculation
    	\end{itemize}
	\end{block}

	\begin{block}{A useful basis set should be\ldots}
		\begin{description}
			\item[\textbf{Accurate}] Flexible enough to describe the relevant
			      occupied and virtual orbitals
			\item[\textbf{Compact}] Able to reach the target accuracy with as
			      few functions as possible
			\item[\textbf{Stable}] Free from severe linear dependencies
			\item[\textbf{Practical}] Compatible with evaluation of
			      multicenter one- and two-electron integrals
		\end{description}
	\end{block}

	\vfill
	\centering
	$\Rightarrow$ \alert{Basis set design is a compromise between accuracy and cost}
\end{frame}

\begin{frame}{Common basis functions}
	\begin{columns}
		\begin{column}{0.5\textwidth}
        	\begin{block}{Gaussian-type orbitals (GTOs)}
				\begin{equation*}
					\phi(\br) = \exp(-\alpha r^2)
    			\end{equation*}
    			\begin{itemize}
    				\item Usually atom-centered
    				\item Standard choice for molecular calculations
    				\item Efficient evaluation of multicenter integrals
    				\item Software: Orca, Q-Chem, Molcas, Molpro, PySCF, Psi4, CFOUR, MRCC, GAMESS, etc
    			\end{itemize}
        	\end{block}
        	\begin{block}{Slater-type orbitals (STOs)} 
				\begin{equation*}
					\phi(\br) = \exp(-\zeta r)
        		\end{equation*}
    			\begin{itemize}
    				\item Usually atom-centered
    				\item Natural nuclear cusp and exponential tail
    				\item Software: ADF
				\end{itemize}		
			\end{block}
		\end{column}
		\begin{column}{0.5\textwidth}
    		\begin{block}{Plane waves (PWs)}
				\begin{equation*}
					\phi(\br) = \exp(\ii \bk \cdot \br)
				\end{equation*}
    			\begin{itemize}
    				\item Delocalized and orthonormal
    				\item Naturally adapted to periodic systems
    				\item Systematically converged with an energy cutoff
    				\item Software: Quantum ESPRESSO, VASP, ABINIT, CASTEP, etc
    			\end{itemize}
			\end{block}
        	\begin{block}{Numerical atomic orbitals (NAOs)}
    			\begin{itemize}
    				\item Radial functions tabulated numerically
    				\item Compact and flexible atom-centered basis
    				\item Software: FHI-aims, SIESTA, etc
    			\end{itemize}
    		\end{block}
			\vspace{0.14\textwidth}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Plane waves}
	\begin{block}{Definition}
		\begin{equation*}
			\phi_{\bk}(\br) = \frac{1}{\sqrt{\Omega}}\exp(\ii \bk \cdot \br)
			\qq{with}
			\abs{\bk} \le k_\text{cut}
			\qq{and} 
			E_\text{cut} = \frac{1}{2} k_\text{cut}^2
		\end{equation*}
	\end{block}
	\begin{block}{Reciprocal-lattice vector}
		\begin{equation*}
			\bk = k_1 \bb_1 + k_2 \bb_2 + k_3 \bb_3 
		\end{equation*}
	\end{block}
	\begin{block}{Unit cell}
		\begin{equation*}
			\underbrace{\Omega}_{\text{volume}} = \ba_1 \cdot (\ba_2 \cross \ba_3)
			\qq{with}
			\bA = (\ba_1,\ba_2,\ba_3)
		\end{equation*}		
	\end{block}
	\begin{block}{Vectors of the reciprocal lattice}
		\begin{align*}
			\bb_1 & = 2\pi \frac{\ba_2 \cross \ba_3}{\Omega}
			&
			\bb_2 & = 2\pi \frac{\ba_3 \cross \ba_1}{\Omega}
			&
			\bb_3 & = 2\pi \frac{\ba_1 \cross \ba_2}{\Omega}
		\end{align*}		
	\end{block}
\end{frame}

\begin{frame}{Integrals in the plane wave basis}
	\begin{block}{Overlap integrals}
		\begin{equation*}
			\braket{\bk_1}{\bk_2} 
			= \int \phi_{\bk_1}^*(\br) \phi_{\bk_2}(\br) \dd{\br} 
			= \frac{1}{\Omega} \int e^{\ii \qty( \bk_2 - \bk_1 ) \cdot \br} \dd{\br} 
			= \delta_{\bk_1 \bk_2}
			\qq{$\Rightarrow$}
			\alert{\text{Orthogonal basis!}}
		\end{equation*}
	\end{block}
	\begin{block}{Kinetic integrals}
		\begin{equation*}
			\mel{\bk_1}{-\frac{\laplacian}{2}}{\bk_2} 
			= \int \phi_{\bk_1}^*(\br) \qty( - \frac{\laplacian}{2}  ) \phi_{\bk_2}(\br) \dd{\br} 
			= \int \phi_{\bk_1}^*(\br) \frac{\bk_2^2}{2} \phi_{\bk_2}(\br) \dd{\br} 
			= \frac{\bk_2^2}{2} \delta_{\bk_1 \bk_2}
			\qq{$\Rightarrow$}
			\alert{\text{Diagonal!}}
		\end{equation*}
	\end{block}
	\begin{block}{Electron repulsion integrals (ERIs)}
		\begin{equation*}
		\begin{split}
			\braket{\bk_1 \bk_2 }{\bk_3 \bk_4} 
			& = \iint \phi_{\bk_1}^*(\br_1) \phi_{\bk_2}^*(\br_2) r_{12}^{-1} \phi_{\bk_3}(\br_1) \phi_{\bk_4}(\br_2) \dd{\br_1} \dd{\br_2} 
			\\
			& = \frac{1}{\Omega^2} \iint \frac{e^{\ii \qty( \bk_3 - \bk_1 ) \cdot \br_1} e^{\ii \qty( \bk_4 - \bk_2 ) \cdot \br_2}}{r_{12}} \dd{\br_1} \dd{\br_2} 
		\end{split}
		\end{equation*}
	\end{block}
\end{frame}

\begin{frame}{Resolution of the Coulomb operator}
	\begin{block}{Coulomb operator}
		\begin{equation*}
			\frac{1}{r_{12}} 
			= \frac{1}{\Omega} \sum_{\bk \neq \bO} \frac{4\pi}{\bk^2} e^{\ii \bk \cdot (\br_1 - \br_2)}
		\end{equation*}
		Dealing with the points where $\bk = \bO$ is a bit tricky bud doable...
	\end{block}
	\begin{block}{Electron repulsion integrals}
		\begin{equation*}
		\begin{split}
			\braket{\bk_1 \bk_2 }{\bk_3 \bk_4} 
			& =  \frac{1}{\Omega^3} \sum_{\bk\neq\bO} \frac{4\pi}{\bk^2} \iint e^{\ii \qty( \bk_3 - \bk_1 - \bk ) \cdot \br_1} e^{\ii \qty( \bk_4 - \bk_2 +\bk) \cdot \br_2} \dd{\br_1} \dd{\br_2} 
			\\
			& =  \frac{1}{\Omega} \sum_{\bk\neq\bO} \frac{4\pi}{\bk^2} \delta_{\bk_3-\bk_1-\bk,\bO} \delta_{\bk_4-\bk_2+\bk,\bO}
			\\
			& = \frac{4\pi}{\Omega \abs{\bk_1 - \bk_3}^2} \delta_{\bk_1+\bk_2,\bk_3+\bk_4}
			\qq{$\Rightarrow$}
			\alert{\text{Momentum conservation!}}
		\end{split}
		\end{equation*}
	\end{block}
\end{frame}

\begin{frame}{Plane waves: strengths and limitations}
	\begin{center}
		\fbox{
		\textbf{Key idea:}
		Assemblies of atoms are slight distortions of free electrons
		}
	\end{center}

	\begin{columns}[T,onlytextwidth]
		%
		\begin{column}{0.485\textwidth}
			\begin{block}{Advantages}
				\begin{itemize}
					\item[$\cmark$] Orthonormal and independent of atomic positions
					\item[$\cmark$] Naturally compatible with periodic boundary conditions
					\item[$\cmark$] Systematic convergence through a single parameter
					      $E_{\mathrm{cut}}$
					\item[$\cmark$] Kinetic and Coulomb operators are diagonal in
					      reciprocal space
					\item[$\cmark$] Efficient evaluation using fast Fourier transforms
					\item[$\cmark$] No basis-set superposition error or
					      atom-centered Pulay forces
				\end{itemize}
			\end{block}
		\end{column}
		%
		\begin{column}{0.485\textwidth}
			\begin{block}{Limitations}
				\begin{itemize}
					\item[$\xmark$] Large basis sets:
					$$N_{\text{PW}} \propto \Omega E_{\text{cut}}^{3/2}$$
					\item[$\xmark$] Core orbitals require prohibitively large cutoffs
					      $\Rightarrow$ pseudopotentials
					\item[$\xmark$] Computational cost increases with supercell size
					\item[$\xmark$] Delocalized basis functions provide little
					      real-space sparsity
					\item[$\xmark$] Large virtual spaces make 
					      correlated methods expensive
				\end{itemize}
			\end{block}
		\end{column}
	\end{columns}
	\vfill
	\centering
	$\Rightarrow$ \alert{Particularly effective for periodic valence-electron calculations}
\end{frame}

\begin{frame}{Slater vs Gaussian}
	\begin{columns}
		\begin{column}[t]{0.5\textwidth}
			\begin{block}{Slater-type orbitals (STOs)}
				\begin{equation*}
					\boxed{\phi_{\zeta n\ell m}^\text{STO}(\br) \propto r^{n-1} e^{-\zeta r} Y_{\ell m}(\hat{\br})}
				\end{equation*}
				\begin{itemize}
    				\item[$\cmark$] It can model the electron-nucleus cusp!
					\begin{equation*}
    					\phi_\text{1s}^\text{STO}(r) \propto e^{-\zeta r}  
    					\qq{$\Rightarrow$}
    					\eval{\dv{\phi_\text{1s}^\text{STO}(r)}{r}}_{r=0} = - \zeta
    				\end{equation*}
    				\item[$\cmark$] Correct long-range behavior for IP!
    				\begin{equation*}
    					\phi_\text{HOMO}^\text{STO}(r) \propto e^{-\sqrt{-2\epsilon_\text{HOMO}}r}  
    					\qq{$\Rightarrow$}
    					\text{IP} = - \epsilon_\text{HOMO}
    				\end{equation*}
					\item[$\xmark$] The multicenter integrals are \alert{hard} to compute!
				\end{itemize}    				
			\end{block}
		\end{column}
		\begin{column}[t]{0.5\textwidth}
			\begin{block}{Gaussian-type orbitals (GTOs)}
				\begin{equation*}
					\boxed{\phi_{\alpha n\ell m}^\text{GTO}(\br) \propto r^{2(n-1)-\ell} e^{-\alpha r^2} Y_{\ell m}(\hat{\br})}
				\end{equation*}
				\begin{itemize}
    				\item[$\xmark$] They are flat at the electron-nucleus coalescence point!
					\begin{equation*}
    					\phi_\text{1s}^\text{GTO}(r) \propto e^{-\alpha r^2}  
    					\qq{$\Rightarrow$}
    					\eval{\dv{\phi_\text{1s}^\text{GTO}(r)}{r}}_{r=0} = 0
    				\end{equation*}			
    				\item[$\xmark$] They decay too rapidly at large $r$!
					\item[$\cmark$] The multicenter integrals are \green{easy} to compute!
				\end{itemize}    				
			\end{block}
		\end{column}
	\end{columns}
	\vfill
	\centering
	$\Rightarrow$ \alert{GTOs usually win but we use more!}
\end{frame}

\begin{frame}{Cartesian vs Spherical Gaussians}
	\begin{columns}
		\begin{column}[t]{0.5\textwidth}
			\begin{block}{Cartesian Gaussians}
				For a shell of angular momentum $\ell$
				\begin{equation*}
					\boxed{N_\ell^\text{cart} = \frac{(\ell+1)(\ell+2)}{2}}
				\end{equation*}
				\begin{equation*}
					\begin{array}{c|cccc}
						\ell & s & p & d & f
						\\
						\hline
						N_\ell^\text{cart} & 1 & 3 & \red{6} & \red{10}
					\end{array}
				\end{equation*}
				For instance, a Cartesian $d$ shell contains
				\begin{equation*}
					\{x^2,y^2,z^2,xy,xz,yz\}
				\end{equation*}
			\end{block}
		\end{column}
		\begin{column}[t]{0.5\textwidth}
			\begin{block}{Spherical Gaussians}
				For a pure angular momentum $\ell$
				\begin{equation*}
					\boxed{N_\ell^\text{sph} = 2\ell+1}
				\end{equation*}
				\begin{equation*}
					\begin{array}{c|cccc}
						\ell & s & p & d & f
						\\
						\hline
						N_\ell^\text{sph} & 1 & 3 & \green{5} & \green{7}
					\end{array}
				\end{equation*}
				The extra Cartesian functions correspond to
				\alert{lower-angular-momentum components}:
				\begin{equation*}
					\red{6d} = \green{5d} + 1s
					\qquad
					\red{10f} = \green{7f} + 3p
				\end{equation*}
			\end{block}
		\end{column}
	\end{columns}
	\vfill
	\begin{equation*}
		x^2+y^2+z^2 = r^2
		\qq{$\Rightarrow$}
		\text{the Cartesian $d$ shell contains an $s$-type component}
	\end{equation*}
	\vfill
	\centering
	$\Rightarrow$ \alert{Most quantum-chemistry calculations use spherical (pure) functions}
\end{frame}

\begin{frame}{The contraction problem}
	\begin{columns}
		\begin{column}{0.7\textwidth}
			\begin{block}{Primitive vs Contracted}
				\begin{itemize}
					\item Same center $\bA$

\begin{frame}{Choose the basis for the physics you need}
	\begin{block}{Basis set choice is tricky}
    	\begin{itemize}
 			\bigskip
   		\item \textbf{Valence flexibility:} use double-, triple-, or higher-$\zeta$ descriptions to give each valence orbital several radial shapes
			\bigskip
    		\item \textbf{Polarization functions:} add \alert{higher angular momenta} to describe anisotropic density, bonding, and angular correlation
			\bigskip
    		\item \textbf{Diffuse functions:} add \alert{small exponents} to describe long-range density
    		\\
    		(anions, Rydberg states, noncovalent interactions, multipoles, (hyper)polarizabilities, etc)
			\bigskip
    		\item \textbf{Tight functions:} add \alert{large exponents} to descrive core region
    		\\
    		(core correlation, nuclear magnetic shielding, spin-spin coupling constants, etc)
			\bigskip
    	\end{itemize}
	\end{block}
	\vfill
	\centering
	$\Rightarrow$ \alert{The appropriate basis depends on both the electronic structure method and the property of interest!}
\end{frame}

\begin{frame}{Basis-set superposition error (BSSE)}
	\begin{block}{Notation}
		We denote by
		\begin{equation*}
			\boxed{
				E_{\red{X}}^{\blue{Y}}
				=
				\text{energy of system $\red{X}$ computed in the basis of system $\blue{Y}$}
			}
		\end{equation*}
		For example, $E_{\red{A}}^{\blue{AB}}$ is the energy of fragment $A$
		computed using the basis functions centered on both $A$ and $B$.
		\\
		The basis functions on $B$ are then \violet{ghost functions}
		(no nucleus and no electrons on $B$).
	\end{block}
	\vfill
	\begin{center}
		\begin{tikzpicture}[
			>=stealth',
			atom/.style={circle,draw,minimum size=0.7cm,inner sep=0pt},
			ghost/.style={circle,draw,dashed,minimum size=0.7cm,inner sep=0pt},
			basis/.style={ellipse,draw,minimum width=1.5cm,minimum height=1.0cm},
			ghostbasis/.style={ellipse,draw,dashed,minimum width=1.5cm,minimum height=1.0cm}
		]
			% ------------------------------------------------------------
			% Monomer A in its own basis
			% ------------------------------------------------------------
			\node at (-4.5,1.6) {$E_{\red{A}}^{\blue{A}}$};
			\node[basis] (basisA1) at (-4.5,0) {};
			\node[atom]  (A1)      at (-4.5,0) {$A$};
			\node at (-4.5,-0.9) {\small basis on $A$ only};

			% ------------------------------------------------------------
			% Monomer A in the dimer basis
			% ------------------------------------------------------------
			\node at (0,1.6) {$E_{\red{A}}^{\blue{AB}}$};
			\node[basis]      (basisA2) at (-0.9,0) {};
			\node[atom]       (A2)      at (-0.9,0) {$A$};
			\node[ghostbasis,violet] (basisB2) at ( 0.9,0) {};
			\node[ghost,violet]      (B2)      at ( 0.9,0) {$B$};
			\node at (0,-0.9) {\small basis on $A$ + \violet{ghost basis} on $B$};

			% ------------------------------------------------------------
			% Dimer AB
			% ------------------------------------------------------------
			\node at (4.9,1.6) {$E_{\red{AB}}^{\blue{AB}}$};
			\node[basis] (basisA3) at (4.0,0) {};
			\node[atom]  (A3)      at (4.0,0) {$A$};
			\node[basis] (basisB3) at (5.8,0) {};
			\node[atom]  (B3)      at (5.8,0) {$B$};
			\node at (4.9,-0.9) {\small full dimer basis};
		\end{tikzpicture}
	\end{center}
	\vfill
	\begin{equation*}
		E_{\red{A}}^{\blue{AB}} \leq E_{\red{A}}^{\blue{A}}
		\qq{$\Rightarrow$}
		\text{fragment $A$ is artificially stabilized by the basis on $B$}
	\end{equation*}
	\vfill
	\centering
	$\Rightarrow$
	\alert{Fragments can ``borrow'' basis functions from each other}
\end{frame}

\begin{frame}{Counterpoise correction}
	\begin{block}{Boys-Bernardi counterpoise correction}
		With a finite atom-centered basis, each fragment can
		\alert{borrow basis functions from the other fragment}
		\begin{equation*}
			E_A^{AB} \leq E_A^A
			\qquad\text{and}\qquad
			E_B^{AB} \leq E_B^B
		\end{equation*}
		The usual interaction energy is
		\begin{equation*}
			\Delta E_\text{int}
			=
			E_{AB}^{AB} - E_A^A - E_B^B
		\end{equation*}
		while the counterpoise-corrected interaction energy is
		\begin{equation*}
			\boxed{
				\Delta E_\text{int}^\text{CP}
				=
				E_{AB}^{AB} - E_A^{AB} - E_B^{AB}
			}
		\end{equation*}
	\end{block}
	\vfill
	\begin{block}{Basis-set superposition error}
		\begin{equation*}
			\boxed{
				\text{BSSE}
				=
				\Delta E_\text{int}^\text{CP}
				-
				\Delta E_\text{int}
				=
				\qty(E_A^A-E_A^{AB})
				+
				\qty(E_B^B-E_B^{AB})
				\geq 0
			}
		\end{equation*}
	\end{block}
	\vfill
	\centering
	$\Rightarrow$
	\alert{BSSE makes the uncorrected interaction energy artificially too attractive}
\end{frame}

\begin{frame}{Effective core potentials (ECPs) or pseudopotentials}
	\begin{center}
		\fbox{
		\textbf{Key idea:}
		Treat only the chemically relevant electrons explicitly
		}
	\end{center}

	\begin{block}{One-electron ECP}
		Replace the core electrons by an
		\alert{effective one-electron potential} acting on the remaining electrons
		\begin{equation*}
			\hh(\br)
			=
			- \frac{\nabla^2}{2}
			- \sum_A \frac{Z_A}{\abs{\br - \bR_{A}}}
			\qq{$\longrightarrow$}
			\boxed{
				\hh^\text{ECP}(\br)
				=
				- \frac{\nabla^2}{2}
				- \sum_A \frac{\green{Q_A}}{\abs{\br - \bR_{A}}}
				+
				\sum_A \alert{\hat{U}_A^\text{ECP}(\br)}
			}
		\end{equation*}
		Here, $\green{Q_A} = Z_A - N_A^\text{core}$ is an \green{effective nuclear charge} and $N_A^\text{core}$ is the number of electrons replaced on atom $A$
	\end{block}

	\begin{block}{Why use ECPs?}
		\begin{itemize}
			\item[$\cmark$] \blue{Lower cost}: fewer electrons and no basis functions
				needed to represent core orbitals
			\item[$\cmark$] \violet{Relativistic effects}: can incorporate scalar-relativistic
				effects and, with a suitable operator, spin-orbit coupling
			\item[$\cmark$] Used in both molecular and periodic calculations,
				with Gaussian or plane-wave basis sets
			\item[$\xmark$] Introduce new \alert{one-electron integrals} that can be tricky to compute
			\item[$\xmark$] Core properties are not described explicitly, and accuracy depends on the ECP quality
		\end{itemize}
	\end{block}
\end{frame}

%-----------------------------------------------------
\begin{frame}{ECPs: projected and unprojected terms}
	\begin{block}{A common semilocal form: scalar ECP}
		\begin{equation*}
			\boxed{
				\red{\hat{U}^\text{ECP}(\br)}
				=
				\violet{U_L(r)}
				+
				\sum_{\ell=0}^{L-1} \sum_{m=-\ell}^{\ell} \ket{Y_{\ell m}} \purple{U_\ell(r)} \bra{Y_{\ell m}}
			}
		\end{equation*}
		\begin{itemize}
			\item $\violet{U_L(r)}$: short-range \blue{local} potential $\Leftrightarrow$ \violet{unprojected part}
			\item $\{\ket{Y_{\ell m}} \purple{U_\ell(r)} \bra{Y_{\ell m}}\}_{0\le\ell<L}$: angular-momentum-dependent \blue{semi-local} potentials $\Leftrightarrow$ \purple{projected part}
		\end{itemize}
	\end{block}

	\begin{block}{Gaussian expansion}
		\begin{equation*}
			U_\ell(r)
			=
			\sum_{k=1}^{K_\ell}
			D_{\ell k}\,r^{n_{\ell k}}
			\exp(-\eta_{\ell k}r^2)
			\qquad \ell=0,\ldots,L
		\end{equation*}
		\begin{itemize}
			\item $D_{\ell k}$ and $\eta_{\ell k}>0$ are contraction coefficients and Gaussian exponents
			\item $n_{\ell k}$: radial powers, often $-2$, $-1$, or $0$
			\item The angular projectors complicate the integrals, but they remain \alert{one-electron integrals}
		\end{itemize}
	\end{block}
\end{frame}

\begin{frame}{Properties of Gaussian functions}
	\begin{block}{Gaussian product rule: \textit{``The product of two gaussians is a gaussian''}}
		\begin{equation*}
			G_{\red{\alpha},\red{\bA}}(\br) = \exp(-\red{\alpha} \abs{\br - \red{\bA}}^2)	
			\qqtext{and}
			G_{\blue{\beta},\blue{\bB}}(\br) = \exp(-\blue{\beta} \abs{\br - \blue{\bB}}^2)	
			\qqtext{then}
		\end{equation*}	
		\begin{equation*}
		\boxed{G_{\red{\alpha},\red{\bA}}(\br) G_{\blue{\beta},\blue{\bB}}(\br) = K_{\violet{\zeta},\violet{\bP}} G_{\violet{\zeta},\violet{\bP}}(\br)}
		\qqtext{with} 
		\violet{\zeta} = \red{\alpha} + \blue{\beta} 
		\qqtext{and} 
		\violet{\bP} = \frac{\red{\alpha \bA} + \blue{\beta \bB}}{\red{\alpha} + \blue{\beta} } 
		\end{equation*}		
		\begin{equation*}
			K_{\violet{\zeta},\violet{\bP}} = \exp( -\frac{\red{\alpha} \blue{\beta}}{\red{\alpha} + \blue{\beta} } \abs{\red{\bA} - \blue{\bB}}^2)
			\qq{$\Rightarrow$}
			\text{decay exponentially fast with $\abs{\red{\bA} - \blue{\bB}}$}
		\end{equation*}		
	\end{block}
	\begin{block}{Overlap}
		\begin{equation*}
			\begin{split}
%			\rbraket{\red{\ba}}{ \blue{\bb} }
			\int G_{\red{\alpha},\red{\bA}}(\br) G_{\blue{\beta},\blue{\bB}}(\br) \dd{\br}
			& = K_{\violet{\zeta},\violet{\bP}} \int G_{\violet{\zeta},\violet{\bP}}(\br) \dd{\br}
			\\
			& = K_{\violet{\zeta},\violet{\bP}} 
			\qty( \int_{-\infty}^{+\infty} e^{-\violet{\zeta} x^2} \dd{x} )
			\qty( \int_{-\infty}^{+\infty} e^{-\violet{\zeta} y^2} \dd{y} )
			\qty( \int_{-\infty}^{+\infty} e^{-\violet{\zeta} z^2} \dd{z} )
			\\
			& = K_{\violet{\zeta},\violet{\bP}} 
			\qty( \frac{\pi}{\violet{\zeta}})^{3/2}
			\qq{$\Rightarrow$}
			\text{\alert{cartesian factorization!}}
		\end{split}	
		\end{equation*}	
	\end{block}
\end{frame}

\begin{frame}{One-electron integrals over Gaussian functions}
    \begin{block}{Kinetic energy}
    	\begin{equation*}
    		-\frac{1}{2}\laplacian G_{\blue{\beta},\blue{\bB}}(\br)
    		=
    		\qty[
    			3\blue{\beta}
    			-
    			2\blue{\beta}^2 \abs{\br-\blue{\bB}}^2
    		]
    		G_{\blue{\beta},\blue{\bB}}(\br)
    	\end{equation*}
    	\begin{equation*}
    		\begin{split}
    %			\mel{\red{\ba}}{-\frac{1}{2}\laplacian}{\blue{\bb}}
    			-\frac{1}{2}
    			\int
    			G_{\red{\alpha},\red{\bA}}(\br)
    			\laplacian
    			G_{\blue{\beta},\blue{\bB}}(\br)
    			\dd{\br}
    			& =
    			\int
    			G_{\red{\alpha},\red{\bA}}(\br)
    			\qty[
    				3\blue{\beta}
    				-
    				2\blue{\beta}^2
    				\abs{\br-\blue{\bB}}^2
    			]
    			G_{\blue{\beta},\blue{\bB}}(\br)
    			\dd{\br}
    			\\
    			& =
    			\frac{\red{\alpha}\blue{\beta}}{\violet{\zeta}}
    			\qty[
    				3
    				-
    				2\frac{\red{\alpha}\blue{\beta}}{\violet{\zeta}}
    				\abs{\red{\bA}-\blue{\bB}}^2
    			]
    			K_{\violet{\zeta},\violet{\bP}}
    			\qty(\frac{\pi}{\violet{\zeta}})^{3/2}
    		\end{split}
    	\end{equation*}
    \end{block}

    \begin{block}{Nuclear attraction}
    	For a nucleus of charge $Z_C$ located at $\bC$
    	\begin{equation*}
    		\begin{split}
    			-Z_C
    			\int
    			G_{\red{\alpha},\red{\bA}}(\br)
    			\frac{1}{\abs{\br-\bC}}
    			G_{\blue{\beta},\blue{\bB}}(\br)
    			\dd{\br}
    			& =
    			-Z_C
    			K_{\violet{\zeta},\violet{\bP}}
    			\int
    			\frac{
    				e^{-\violet{\zeta}\abs{\br-\violet{\bP}}^2}
    			}{
    				\abs{\br-\bC}
    			}
    			\dd{\br}
    			\\
    			& =
    			-\frac{2Z_C}{\sqrt{\pi}}
    			K_{\violet{\zeta},\violet{\bP}}
    			\int_0^\infty
    			\dd{t}
    			\int
    			e^{-\violet{\zeta}\abs{\br-\violet{\bP}}^2}
    			e^{-t^2\abs{\br-\bC}^2}
    			\dd{\br}
    			\\
    			& =
    			-\frac{2\pi Z_C}{\violet{\zeta}}
    			K_{\violet{\zeta},\violet{\bP}}
    			F_0\qty(
    				\violet{\zeta}
    				\abs{\violet{\bP}-\bC}^2
    			)
    		\end{split}
    	\end{equation*}
    \end{block}
\end{frame}

\begin{frame}{Boys function}
	\begin{block}{Generalized Boys function}
		\begin{equation*}
			F_m(x) = \int_0^1 t^{2m} \exp(-x t^2) \dd{t}
			\qquad
			F_0(x) = \sqrt{\frac{\pi}{4x}} \erf(\sqrt{x})
		\end{equation*}
	\end{block}
	\begin{block}{Computation}
		\begin{itemize}
			\item $F_m(x)$ computed using pretabulated values on a grid and Taylor expansion (Chebyshev interpolation table)
			\item higher $m$ values computed via downward recursion
		\end{itemize}
	\end{block}
	\begin{block}{Downward recursion}
		\begin{equation*}
			F_m(x) = \frac{2x F_{m+1} (x) + \exp(-x)}{2m+1}
		\end{equation*}
		Downward recursion is numerically stable, whereas upward recursion can become unstable for small $x$
	\end{block}
	\begin{block}{Asymptotic behavior}
		\begin{equation*}
			F_m(x) \sim \frac{(2m-1)!!}{2^{m+1}} \sqrt{\frac{\pi}{x^{2m+1}}}
			\qq{for large $x$}
		\end{equation*}
	\end{block}
\end{frame}

\begin{frame}{Auxiliary fundamental integrals}

\begin{block}{Why do higher-order Boys functions appear?}

The physical $ssss$ ERI involves only $F_0$.

However, generating angular momentum by differentiation produces
higher-order Boys functions $F_m$.

It is therefore convenient to define the hierarchy

\begin{equation*}
\boxed{
\sbraket{\bO\bO}{\bO\bO}^{(m)}
=
\sqrt{\frac{4\vartheta}{\pi}}
\qty(\frac{\pi}{\violet{\zeta}})^{3/2}
K_{\violet{\zeta},\violet{\bP}}
\qty(\frac{\pi}{\purple{\eta}})^{3/2}
K_{\purple{\eta},\purple{\bQ}}
F_m(\vartheta \abs{\violet{\bP} - \purple{\bQ}}^2)
}
\end{equation*}

\end{block}

\begin{itemize}
	\item $m=0$: \alert{physical} $ssss$ ERI
	\item $m>0$: \alert{auxiliary integrals}, used only as intermediates
	\item Angular-momentum recurrences couple auxiliary orders $m$ and $m+1$
	\item For total angular momentum $L=a+b+c+d$
	\\
	$\Rightarrow$ one needs fundamental auxiliary integrals up to $m=L$.
\end{itemize}

\end{frame}

\begin{frame}{Obara-Saika: generating angular momentum}

	\begin{block}{Key identity}
		For a Cartesian Gaussian,
		\begin{equation*}
			\pdv{\sket{\red{\ba}}}{A_i}
            =
            2\alpha
            \sket{\red{\ba}+\bI_i}
            -
            a_i
            \sket{\red{\ba}-\bI_i}
            \qq{$\Rightarrow$}
            \boxed{
            \sket{\red{\ba}+\bI_i}
            =
            \frac{1}{2\alpha}
            \qty[
            \pdv{\sket{\red{\ba}}}{A_i}
            +
            a_i\sket{\red{\ba}-\bI_i}
            ]
            }
		\end{equation*}
	\end{block}

	Applying this identity to the auxiliary ERIs generates recurrence relations connecting different angular momenta and different Boys orders
	$\Rightarrow$ \alert{Obara-Saika or vertical recurrence relations}

    \vfill
    
    \begin{equation*}
    \boxed{
    \sbraket{00}{00}^{(m)}
    \quad\longrightarrow\quad
    \sbraket{\red{a}0}{00}^{(m)}
    \quad\longrightarrow\quad
    \sbraket{\red{a}0}{\orange{c}0}^{(m)}
    }
    \end{equation*}
    
    \vfill
    
    \centering
    \alert{Instead of evaluating every ERI from scratch, we build them recursively}

\end{frame}

\begin{frame}{Strategy for building an ERI}

	Starting from the fundamental auxiliary integrals,
    \begin{equation*}
        \boxed{
        \sbraket{00}{00}^{(m)}
        \;
        \xrightarrow{\;\red{\mathrm{VRR}}\;}
        \;
        \sbraket{\red{a}0}{\orange{c}0}^{(m)}
        \;
        \xrightarrow{\;\orange{\mathrm{contraction}}\;}
        \;
        \rbraket{\red{a}0}{\orange{c}0}
        \;
        \xrightarrow{\;\blue{\mathrm{HRR}}\;}
        \;
        \rbraket{\red{a}\blue{b}}{\orange{c}\green{d}}
        }
    \end{equation*}

	\vfill
    \begin{itemize}
    	\item \red{Vertical recurrence relations (VRRs)}: create angular momentum on one center of each Gaussian pair
    	\item \orange{Contraction}: combine primitive integrals into contracted integrals
    	\item \blue{Horizontal recurrence relations (HRRs)}: redistribute angular momentum within each Gaussian pair
    \end{itemize}

	\vfill

    \begin{equation*}
        \boxed{
        \text{\red{VRRs create angular momentum}
        \qquad
        \text{\blue{HRRs redistribute it}}}
        }
    \end{equation*}

\end{frame}


% ======================================================================
% VRR on A
% ======================================================================

\begin{frame}{Vertical recurrence relations}

\small

\begin{block}{VRR on the bra pair: build angular momentum on $\bA$}

\begin{equation*}
\begin{split}
\sbraket{(\red{\ba}+\bI_i)\bO}{\bO\bO}^{(m)}
={}&
(P_i-A_i)
\sbraket{\red{\ba}\bO}{\bO\bO}^{(m)}
\\
&+
\frac{\eta}{\zeta+\eta}
(P_i-Q_i)
\sbraket{\red{\ba}\bO}{\bO\bO}^{(m+1)}
\\
&+
\frac{a_i}{2\zeta}
\qty[
\sbraket{(\red{\ba}-\bI_i)\bO}{\bO\bO}^{(m)}
-
\frac{\eta}{\zeta+\eta}
\sbraket{(\red{\ba}-\bI_i)\bO}{\bO\bO}^{(m+1)}
]
\end{split}
\end{equation*}

\end{block}


\begin{block}{VRR on the ket pair: build angular momentum on $\bC$}

\begin{equation*}
\begin{split}
\sbraket{\red{\ba}\bO}{(\orange{\bc}+\bI_i)\bO}^{(m)}
={}&
(Q_i-C_i)
\sbraket{\red{\ba}\bO}{\orange{\bc}\bO}^{(m)}
\\
&+
\frac{\zeta}{\zeta+\eta}
(Q_i-P_i)
\sbraket{\red{\ba}\bO}{\orange{\bc}\bO}^{(m+1)}
\\
&+
\frac{c_i}{2\eta}
\qty[
\sbraket{\red{\ba}\bO}{(\orange{\bc}-\bI_i)\bO}^{(m)}
-
\frac{\zeta}{\zeta+\eta}
\sbraket{\red{\ba}\bO}{(\orange{\bc}-\bI_i)\bO}^{(m+1)}
]
\\
&+
\frac{a_i}{2(\zeta+\eta)}
\sbraket{(\red{\ba}-\bI_i)\bO}{\orange{\bc}\bO}^{(m+1)}
\end{split}
\end{equation*}


\end{block}

\end{frame}


% ======================================================================
% HRR
% ======================================================================

\begin{frame}{Horizontal recurrence relations}

\begin{block}{Redistributing angular momentum}

Once angular momentum has been generated on one center of a Gaussian
pair, it can be shifted to the other center:

\begin{equation*}
\boxed{
\sbraket{\red{\ba}(\blue{\bb}+\bI_i)}
        {\orange{\bc}\green{\bd}}
=
\sbraket{(\red{\ba}+\bI_i)\blue{\bb}}
        {\orange{\bc}\green{\bd}}
+
(A_i-B_i)
\sbraket{\red{\ba}\blue{\bb}}
        {\orange{\bc}\green{\bd}}
}
\end{equation*}

and, similarly, angular momentum can be transferred from $\bC$ to $\bD$

\end{block}

\begin{block}{Why apply HRRs after contraction?}

The HRR depends only on the geometry through $A_i-B_i$:
it is \alert{independent of the Gaussian exponents}

Therefore, the same recurrence applies directly to contracted integrals

\begin{equation*}
\boxed{
\rbraket{\red{\ba}(\blue{\bb}+\bI_i)}
        {\orange{\bc}\green{\bd}}
=
\rbraket{(\red{\ba}+\bI_i)\blue{\bb}}
        {\orange{\bc}\green{\bd}}
+
(A_i-B_i)
\rbraket{\red{\ba}\blue{\bb}}
        {\orange{\bc}\green{\bd}}
}
\end{equation*}

\end{block}

\end{frame}

\begin{frame}{Late-contraction ERI algorithm}

    \begin{equation*}
        \boxed{
        \underbrace{
        \sbraket{00}{00}^{(m)}
        }_{\substack{\text{fundamental}\\\text{auxiliary integrals}}}
        \quad
        \xrightarrow{\;\red{\mathrm{VRR}}\;}
        \quad
        \underbrace{
        \sbraket{\red{a}0}{\orange{c}0}
        }_{\text{primitive integrals}}
        \quad
        \xrightarrow{\;\orange{\mathrm{contract}}\;}
        \quad
        \underbrace{
        \rbraket{\red{a}0}{\orange{c}0}
        }_{\text{contracted integrals}}
        \quad
        \xrightarrow{\;\blue{\mathrm{HRR}}\;}
        \quad
        \underbrace{
        \rbraket{\red{a}\blue{b}}{\orange{c}\green{d}}
        }_{\text{target ERIs}}
        }
    \end{equation*}

	\vfill

    \begin{enumerate}
    	\item Compute the required Boys functions $F_m(T)$
    	\item Form the fundamental auxiliary integrals $\sbraket{00}{00}^{(m)}$
    	\item Use \red{VRRs} to build angular momentum on $\bA$ and $\bC$
    	\item \orange{Contract} the primitive integrals
    	\item Use exponent-independent \blue{HRRs} to transfer angular momentum $\bA \rightarrow \bB$ and $\bC \rightarrow \bD$
    \end{enumerate}

	\vfill

	\centering
	\alert{Expensive exponent-dependent recurrences are performed before contraction, while the cheap exponent-independent HRRs are performed afterwards}

\end{frame}




\begin{frame}{Upper bounds for ERIs}
	\begin{block}{A ``good'' upper bound must be}
		\begin{itemize}
			\item simple, i.e, cheap to compute
			\item strong, i.e., a good estimate
			\item scaling-consistent, i.e., $N_\text{sig} = \order*{N_\text{upper}}$
		\end{itemize}
	\end{block}
	\begin{block}{Cauchy-Schwartz bound}
		\begin{equation*}
			\abs{\rbraket{ \red{\ba} \blue{\bb} }{ \orange{\bc} \green{\bd} }} 
			\le 
			\sqrt{\rbraket{ \red{\ba} \blue{\bb} }{ \red{\ba} \blue{\bb} }} 
			\sqrt{\rbraket{ \orange{\bc} \green{\bd} }{ \orange{\bc} \green{\bd} }}
			\qqtext{or}
			\abs{\rbraket{\violet{\bP}}{\purple{\bQ}}} 
			\le 
			\sqrt{\rbraket{\violet{\bP}}{\violet{\bP}}} 
			\sqrt{\rbraket{\purple{\bQ}}{\purple{\bQ}}}
		\end{equation*}
    	\begin{itemize}
    		\item $\rbraket{\violet{\bP}}{\purple{\bQ}}$ should decay as $1/\abs{\violet{\bP} - \purple{\bQ}}$ or faster
    		\item However, there is no distance information in the product $\rbraket{\violet{\bP}}{\violet{\bP}} \rbraket{\purple{\bQ}}{\purple{\bQ}}$
    		\item The Cauchy-Schwarz bound is rigorous and cheap but is not tight for well-separated shell pairs
    	\end{itemize}
	\end{block}

\begin{frame}{Screening ERIs}

	\begin{block}{Integral bound}
		We seek a single number that bounds a particular integral
		\begin{equation*}
			\abs{\rbraket{ \red{\ba} \blue{\bb} }{ \orange{\bc} \green{\bd} }} 
			\le 
			\sqrt{\rbraket{ \red{\ba} \blue{\bb} }{ \red{\ba} \blue{\bb} }} 
			\sqrt{\rbraket{ \orange{\bc} \green{\bd} }{ \orange{\bc} \green{\bd} }}
		\end{equation*}
		With $\order{K^{2}}$ $\rbraket{ \red{\ba} \blue{\bb} }{ \red{\ba} \blue{\bb} }$ bound factors, we can upper bound $\order{K^{4}}$ $\rbraket{ \red{\ba} \blue{\bb} }{ \orange{\bc} \green{\bd} }$ integrals
	\end{block}

	\begin{block}{Class bound}
		We seek a single number that bounds an entire class
		\begin{equation*}
			\abs{\rbraket{ \red{a} \blue{b} }{ \orange{c} \green{d} }} 
			\le Q_{\red{a} \blue{b}} Q_{\orange{c} \green{d}}
			\qq{with}
			Q_{\red{a} \blue{b}}
			=
			\max_{\substack{\red{\ba}\in a \\ \blue{\bb}\in b}}
			\sqrt{
				\rbraket{\red{\ba}\blue{\bb}}{\red{\ba}\blue{\bb}}
			}
		\end{equation*}
		Therefore,
		\begin{equation*}
			\qif* Q_{\red{a} \blue{b}} Q_{\orange{c} \green{d}} < \tau
			\qthen
			\text{\alert{discard the entire shell quartet!}}
		\end{equation*}
	\end{block}
	*{\small The simple $\rbraket{dd}{dd}$ class is made of 1296 integrals!}
\end{frame}

\begin{frame}{Direct SCF algorithm}
	\begin{block}{From ERIs to the Fock matrix}
		The Coulomb and exchange matrices are
		\begin{align*}
			J_{\mu\nu}
			& =
			\sum_{\lambda\sigma}
			P_{\lambda\sigma}
			\rbraket{\mu\nu}{\lambda\sigma}
			&
			K_{\mu\nu}
			& =
			\sum_{\lambda\sigma}
			P_{\lambda\sigma}
			\rbraket{\mu\lambda}{\nu\sigma}
		\end{align*}
	\end{block}
	\vfill
	\begin{block}{Direct SCF: trading disk storage for repeated integral evaluation}
		Instead of storing all $\order{K^4}$ ERIs
		\begin{enumerate}
			\item loop over shell quartets $\rbraket{\mu\nu}{\la\si}$
			\item use an upper bound to discard negligible quartets
			\item compute the significant ERIs
			\item immediately accumulate their contributions to $\bJ$ and $\bK$
			\item discard the ERIs
		\end{enumerate}
		\alert{The density matrix elements can be included in the screening process}
	\end{block}
	\vfill
	\begin{equation*}
		\boxed{
			\text{screen}
			\;\longrightarrow\;
			\text{compute}
			\;\longrightarrow\;
			\text{accumulate}
			\;\longrightarrow\;
			\text{discard}
		}
	\end{equation*}
\end{frame}

\begin{frame}{Resolution of the identity (RI)}
	\begin{block}{The product space is the problem}
		An ERI is the Coulomb interaction between two \alert{pair densities},
		\begin{equation*}
			\rbraket{\mu\nu}{\lambda\sigma}
			=
			\iint
			\underbrace{\phi_\mu(\br_1)\phi_\nu(\br_1)}_{\rho_{\mu\nu}(\br_1)}
			\frac{1}{r_{12}}
			\underbrace{\phi_\lambda(\br_2)\phi_\sigma(\br_2)}_{\rho_{\lambda\sigma}(\br_2)}
			\dd{\br_1}\dd{\br_2}
		\end{equation*}
		With $K$ basis functions, there are formally $\order{K^2}$
		different pair densities $\rho_{\mu\nu}$
	\end{block}
	\vfill
	\begin{block}{Density fitting}
		Instead of representing each product $\phi_\mu\phi_\nu$ exactly,
		we expand it in a smaller \alert{auxiliary basis} $\{\varphi_A^\text{aux}\}$,
		\begin{equation*}
			\boxed{
				\rho_{\mu\nu}(\br)
				\approx
				\widetilde{\rho}_{\mu\nu}(\br)
				=
				\sum_A C_{\mu\nu}^{A}
				\varphi_A^\text{aux}(\br)
			}
		\end{equation*}
		with typically $K_\text{aux} = \order{K}$
	\end{block}
	\vfill
	\centering
	$\Rightarrow$
	\alert{RI replaces the large AO-product space by a compact auxiliary space}
\end{frame}


% ======================================================================
% Coulomb metric
% ======================================================================

\begin{frame}{RI in the Coulomb metric}
	\begin{block}{How should the pair density be fitted?}
		Define the Coulomb inner product
		\begin{equation*}
			\rbraket{f}{g}
			=
			\iint
			f(\br_1)
			\frac{1}{r_{12}}
			g(\br_2)
			\dd{\br_1}\dd{\br_2}
		\end{equation*}
		The optimal coefficients minimize the Coulomb norm of the fitting error,
		\begin{equation*}
			\Delta_{\mu\nu}
			=
			\rbraket{
				\rho_{\mu\nu}-\widetilde{\rho}_{\mu\nu}
				}{
				\rho_{\mu\nu}-\widetilde{\rho}_{\mu\nu}
			}
		\end{equation*}
	\end{block}
	\vfill
	\begin{block}{Least-squares equations}
		Minimization with respect to $C_{\mu\nu}^{A}$ gives
		\begin{equation*}
			\sum_B
			\underbrace{\rbraket{A}{B}}_{V_{AB}}
			C_{\mu\nu}^{B}
			=
			\underbrace{\rbraket{A}{\mu\nu}}_{\text{three-center integral}}
			\qq{$\Rightarrow$}
			\boxed{
				C_{\mu\nu}^{A}
				=
				\sum_B
				\qty(\bV^{-1})_{AB}
				\rbraket{B}{\mu\nu}
			}
		\end{equation*}
	\end{block}
	\vfill
	\centering
	$\Rightarrow$
	\alert{The Coulomb metric determines the optimal auxiliary expansion}
\end{frame}

\begin{frame}{RI approximation to the ERIs}
	\begin{block}{From four-center to three-center integrals}
		Inserting the fitted pair densities gives
		\begin{equation*}
			\boxed{
				\rbraket{\mu\nu}{\lambda\sigma}
				\approx
				\sum_{AB}
				\rbraket{\mu\nu}{A}
				\qty(\bV^{-1})_{AB}
				\rbraket{B}{\lambda\sigma}
			}
		\end{equation*}
	\end{block}
	\vfill
	\begin{block}{Symmetric factorization}
		Defining the orthonormalized three-index quantities
		\begin{equation*}
			B_{\mu\nu}^{P}
			=
			\sum_A
			\rbraket{\mu\nu}{A}
			\qty(\bV^{-1/2})_{AP}
		\end{equation*}
		one obtains the particularly simple form
		\begin{equation*}
			\boxed{
				\rbraket{\mu\nu}{\lambda\sigma}
				\approx
				\sum_P
				B_{\mu\nu}^{P}
				B_{\lambda\sigma}^{P}
			}
		\end{equation*}
	\end{block}
	\vfill
	\centering
	$\Rightarrow$
	\alert{A four-index tensor is factorized into products of three-index tensors}
\end{frame}

\begin{frame}{Low-rank representations of the ERIs}
	\begin{block}{Resolution of the identity = Density fitting}
		In RI/DF, the AO pair densities are expanded in a
		\alert{predefined auxiliary basis} $\{\chi_A^\text{aux}\}$,
		\begin{equation*}
			\chi_\mu(\br)\chi_\nu(\br)
			\approx
			\sum_A C_{\mu\nu}^A \chi_A^\text{aux}(\br)
			\qq{$\Rightarrow$}
			\boxed{
				\rbraket{\mu\nu}{\lambda\sigma}
				\approx
				\sum_P
				B_{\mu\nu}^P B_{\lambda\sigma}^P
			}
		\end{equation*}
		$\Rightarrow$ \alert{The quality of the approximation depends on the auxiliary basis}
	\end{block}
	\vfill
	\begin{block}{Cholesky decomposition (CD)}
		The ERI tensor $\bV_{(\mu\nu),(\lambda\sigma)} = \rbraket{\mu\nu}{\lambda\sigma}$ can be viewed as a positive semidefinite matrix
		in the compound indices $(\mu\nu)$ and $(\lambda\sigma)$.
		A pivoted Cholesky decomposition directly generates
		\begin{equation*}
			\boxed{
				\rbraket{\mu\nu}{\lambda\sigma}
				\approx
				\sum_{P=1}^{N_\text{CD}}
				L_{\mu\nu}^{P} L_{\lambda\sigma}^{P}
			}
		\end{equation*}
		until the remaining error is smaller than a prescribed threshold $\tau_\text{CD}$
	\end{block}
	\vfill
	\centering
	$\Rightarrow$
	\alert{RI/DF uses a predefined auxiliary space; CD constructs one adaptively}
\end{frame}

\begin{frame}{RI in practice}
	\begin{block}{Construction of the Coulomb matrix}
		Starting from
		\begin{equation*}
			J_{\mu\nu}
			=
			\sum_{\lambda\sigma}
			P_{\lambda\sigma}
			\rbraket{\mu\nu}{\lambda\sigma},
		\end{equation*}
		the RI approximation gives
		\begin{equation*}
			J_{\mu\nu}
			\approx
			\sum_{AB}
			\rbraket{\mu\nu}{A}
			\qty(\bV^{-1})_{AB}
			\underbrace{
				\sum_{\lambda\sigma}
				P_{\lambda\sigma}
				\rbraket{B}{\lambda\sigma}
			}_{d_B}.
		\end{equation*}
		Thus,
		\begin{equation*}
			\boxed{
				\bP
				\;\longrightarrow\;
				\{d_A\}
				\;\longrightarrow\;
				\{c_A\}=\bV^{-1}\{d_A\}
				\;\longrightarrow\;
				\bJ
			}
		\end{equation*}
	\end{block}
	\vfill
	\begin{itemize}
		\item No need to store the full $\order{K^4}$ ERI tensor
		\item Only \alert{two-center} $\rbraket{A}{B}$ and \alert{three-center} $\rbraket{\mu\nu}{A}$ integrals are required
		\item Auxiliary basis sets are optimized to make the fitting error very small
		\item RI is also commonly called \alert{density fitting (DF)}
	\end{itemize}
	\vfill
	\centering
	$\Rightarrow$
	\alert{RI dramatically reduces the cost and memory requirements of Coulomb contractions}
\end{frame}
