\begin{frame}{Expanding the molecular orbitals}
	\begin{equation}
		\boxed{
			\psi_p(\br)
			=
			\sum_{\mu=1}^{K}
			C_{\mu p}\phi_\mu(\br)
		}
	\end{equation}

	\begin{itemize}
		\item The basis functions $\{\phi_\mu\}$ are \alert{fixed}; the coefficients
		      $C_{\mu p}$ are determined \alert{variationally}

		\item Increasing $K$ improves the flexibility of the orbital expansion
		      and, ideally, approaches the \alert{complete basis set limit}

		\item Basis sets contain the \alert{only empirical parameters} of an \textit{ab initio} 
		      calculation
	\end{itemize}

	\begin{block}{A useful basis set should be\ldots}
		\begin{description}
			\item[\textbf{Accurate}] Flexible enough to describe the relevant
			      occupied and virtual orbitals
			\item[\textbf{Compact}] Able to reach the target accuracy with as
			      few functions as possible
			\item[\textbf{Stable}] Free from severe linear dependencies
			\item[\textbf{Efficient}] Compatible with practical evaluation of
			      multicenter one- and two-electron integrals
		\end{description}
	\end{block}

	\vfill
	\centering
	$\Rightarrow$ \alert{Basis set design is a compromise between accuracy and cost}
\end{frame}

\begin{frame}{Common basis functions}
	\begin{columns}
		\begin{column}{0.5\textwidth}
        	\begin{block}{Gaussian-type orbitals (GTOs)}
				\begin{equation}
					\phi(\br) = \exp(-\alpha r^2)
    			\end{equation}
    			\begin{itemize}
    				\item Usually atom-centered
    				\item Standard choice for molecular calculations
    				\item Efficient evaluation of multicenter integrals
    				\item Software: Orca, Q-Chem, Molcas, Molpro, PySCF, Psi4, CFOUR, MRCC, GAMESS, etc
    			\end{itemize}
        	\end{block}
        	\begin{block}{Slater-type orbitals (STOs)} 
				\begin{equation}
					\phi(\br) = \exp(-\zeta r)
        		\end{equation}
    			\begin{itemize}
    				\item Usually atom-centered
    				\item Natural nuclear cusp and exponential tail
    				\item Software: ADF
				\end{itemize}		
			\end{block}
		\end{column}
		\begin{column}{0.5\textwidth}
    		\begin{block}{Plane waves (PWs)}
				\begin{equation}
					\phi(\br) = \exp(\ii \bk \cdot \br)
				\end{equation}
    			\begin{itemize}
    				\item Delocalized and orthonormal
    				\item Naturally adapted to periodic systems
    				\item Systematically converged with an energy cutoff
    				\item Software: Quantum ESPRESSO, VASP, ABINIT, CASTEP, etc
    			\end{itemize}
			\end{block}
        	\begin{block}{Numerical atomic orbitals (NAOs)}
    			\begin{itemize}
    				\item Radial functions tabulated numerically
    				\item Compact and flexible atom-centered basis
    				\item Software: FHI-aims, SIESTA, etc
    			\end{itemize}
    		\end{block}
			\vspace{0.14\textwidth}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Plane waves}
	\begin{block}{Definition}
		\begin{equation}
			\phi_{\bk}(\br) = \frac{1}{\sqrt{\Omega}}\exp(\ii \bk \cdot \br)
			\qq{with}
			\abs{\bk} \le k_\text{cut}
			\qq{and} 
			E_\text{cut} = \frac{1}{2} k_\text{cut}^2
		\end{equation}
	\end{block}
	\begin{block}{Bloch wave vector}
		\begin{equation}
			\bk = k_1 \bb_1 + k_2 \bb_2 + k_3 \bb_3 
		\end{equation}
	\end{block}
	\begin{block}{Unit cell}
		\begin{equation}
			\underbrace{\Omega}_{\text{volume}} = \ba_1 \cdot (\ba_2 \cross \ba_3)
			\qq{with}
			\bA = (\ba_1,\ba_2,\ba_3)
		\end{equation}		
	\end{block}
	\begin{block}{Vectors of the reciprocal lattice}
		\begin{align}
			\bb_1 & = 2\pi \frac{\ba_2 \cross \ba_3}{\Omega}
			&
			\bb_2 & = 2\pi \frac{\ba_3 \cross \ba_1}{\Omega}
			&
			\bb_3 & = 2\pi \frac{\ba_1 \cross \ba_2}{\Omega}
		\end{align}		
	\end{block}
\end{frame}

\begin{frame}{Integrals in the plane wave basis}
	\begin{block}{Overlap integrals}
		\begin{equation}
			\braket{\bk_1}{\bk_2} 
			= \int \phi_{\bk_1}^*(\br) \phi_{\bk_2}(\br) \dd{\br} 
			= \frac{1}{\Omega} \int e^{\ii \qty( \bk_2 - \bk_1 ) \cdot \br} \dd{\br} 
			= \delta_{\bk_1 \bk_2}
			\qq{$\Rightarrow$}
			\alert{\text{Orthogonal basis!}}
		\end{equation}
	\end{block}
	\begin{block}{Kinetic integrals}
		\begin{equation}
			\mel{\bk_1}{-\frac{\laplacian}{2}}{\bk_2} 
			= \int \phi_{\bk_1}^*(\br) \qty( - \frac{\laplacian}{2}  ) \phi_{\bk_2}(\br) \dd{\br} 
			= \int \phi_{\bk_1}^*(\br) \bk_2^2 \phi_{\bk_2}(\br) \dd{\br} 
			= \bk_2^2 \delta_{\bk_1 \bk_2}
			\qq{$\Rightarrow$}
			\alert{\text{Diagonal!}}
		\end{equation}
	\end{block}
	\begin{block}{Electron repulsion integrals}
		\begin{equation}
		\begin{split}
			\braket{\bk_1 \bk_2 }{\bk_3 \bk_4} 
			& = \iint \phi_{\bk_1}^*(\br_1) \phi_{\bk_2}^*(\br_2) r_{12}^{-1} \phi_{\bk_3}(\br_1) \phi_{\bk_4}(\br_2) \dd{\br_1} \dd{\br_2} 
			\\
			& = \frac{1}{\Omega^2} \iint \frac{e^{\ii \qty( \bk_3 - \bk_1 ) \cdot \br_1} e^{\ii \qty( \bk_4 - \bk_2 ) \cdot \br_2}}{r_{12}} \dd{\br_1} \dd{\br_2} 
		\end{split}
		\end{equation}
	\end{block}
\end{frame}

\begin{frame}{Resolution of the Coulomb operator}
	\begin{block}{Coulomb operator}
		\begin{equation}
			\frac{1}{r_{12}} 
			= \frac{1}{\Omega} \sum_{\bk \neq \bO} \frac{4\pi}{\bk^2} e^{\ii \bk \cdot (\br_1 - \br_2)}
		\end{equation}
		Dealing with the points where $\bk = \bO$ is a bit tricky bud doable...
	\end{block}
	\begin{block}{Electron repulsion integrals}
		\begin{equation}
		\begin{split}
			\braket{\bk_1 \bk_2 }{\bk_3 \bk_3} 
			& =  \frac{1}{\Omega^3} \sum_{\bk\neq\bO} \frac{4\pi}{\bk^2} \iint e^{\ii \qty( \bk_3 - \bk_1 - \bk ) \cdot \br_1} e^{\ii \qty( \bk_4 - \bk_2 +\bk) \cdot \br_2} \dd{\br_1} \dd{\br_2} 
			\\
			& =  \frac{1}{\Omega} \sum_{\bk\neq\bO} \frac{4\pi}{\bk^2} \delta_{\bk_3-\bk_1-\bk,\bO} \delta_{\bk_4-\bk_2+\bk,\bO}
			\\
			& = \frac{4\pi}{\Omega \abs{\bk_1 - \bk_3}^2} \delta_{\bk_1+\bk_2,\bk_3+\bk_4}
			\qq{$\Rightarrow$}
			\alert{\text{Momentum conservation!}}
		\end{split}
		\end{equation}
	\end{block}
\end{frame}

\begin{frame}{Plane waves: strengths and limitations}
	\begin{center}
		\fbox{
		\textbf{Key idea:}
		Assemblies of atoms are slight distortions of free electrons
		}
	\end{center}

	\begin{columns}[T,onlytextwidth]
		%
		\begin{column}{0.485\textwidth}
			\begin{block}{Advantages}
				\begin{itemize}
					\item[$\cmark$] Orthonormal and independent of atomic positions
					\item[$\cmark$] Naturally compatible with periodic boundary conditions
					\item[$\cmark$] Systematic convergence through a single parameter
					      $E_{\mathrm{cut}}$
					\item[$\cmark$] Kinetic and Coulomb operators are diagonal in
					      reciprocal space
					\item[$\cmark$] Efficient evaluation using fast Fourier transforms
					\item[$\cmark$] No basis-set superposition error or
					      atom-centered Pulay forces
				\end{itemize}
			\end{block}
		\end{column}
		%
		\begin{column}{0.485\textwidth}
			\begin{block}{Limitations}
				\begin{itemize}
					\item[$\xmark$] Large basis sets:
					$$N_{\text{PW}} \propto \Omega E_{\text{cut}}^{3/2}$$
					\item[$\xmark$] Core orbitals require prohibitively large cutoffs
					      $\Rightarrow$ pseudopotentials
					\item[$\xmark$] Computational cost increases with supercell size
					\item[$\xmark$] Delocalized basis functions provide little
					      real-space sparsity
					\item[$\xmark$] Large virtual spaces make 
					      post-Hartree--Fock methods expensive
				\end{itemize}
			\end{block}
		\end{column}
	\end{columns}
	\vfill
	\centering
	$\Rightarrow$ \alert{Particularly effective for periodic valence-electron calculations}
\end{frame}

\begin{frame}{Slater vs Gaussian}
	\begin{columns}
		\begin{column}[t]{0.5\textwidth}
			\begin{block}{Slater-type orbitals (STOs)}
				\begin{equation}
					\boxed{\phi_{\zeta n\ell m}^\text{STO}(\br) \propto r^{n-1} e^{-\zeta r} Y_{\ell m}(\hat{\br})}
				\end{equation}
				\begin{itemize}
    				\item[$\cmark$] It can model the electron-nucleus cusp!
					\begin{equation}
    					\phi_\text{1s}^\text{STO}(r) \propto e^{-\zeta r}  
    					\qq{$\Rightarrow$}
    					\eval{\dv{\phi_\text{1s}^\text{STO}(r)}{r}}_{r=0} = - \zeta
    				\end{equation}
    				\item[$\cmark$] Correct long-range behavior for IP!
    				\begin{equation}
    					\phi_\text{HOMO}^\text{STO}(r) \propto e^{-\sqrt{-2\epsilon_\text{HOMO}}r}  
    					\qq{$\Rightarrow$}
    					\text{IP} = - \epsilon_\text{HOMO}
    				\end{equation}
					\item[$\xmark$] The multicenter integrals are \alert{hard} to compute!
				\end{itemize}    				
			\end{block}
		\end{column}
		\begin{column}[t]{0.5\textwidth}
			\begin{block}{Gaussian-type orbitals (GTOs)}
				\begin{equation}
					\boxed{\phi_{\alpha n\ell m}^\text{GTO}(\br) \propto r^{2(n-1)-\ell} e^{-\alpha r^2} Y_{\ell m}(\hat{\br})}
				\end{equation}
				\begin{itemize}
    				\item[$\xmark$] They are flat at the electron-nucleus coalescence point!
					\begin{equation}
    					\phi_\text{1s}^\text{GTO}(r) \propto e^{-\alpha r^2}  
    					\qq{$\Rightarrow$}
    					\eval{\dv{\phi_\text{1s}^\text{GTO}(r)}{r}}_{r=0} = 0
    				\end{equation}			
    				\item[$\xmark$] They decay too rapidly at large $r$!
					\item[$\cmark$] The multicenter integrals are \green{easy} to compute!
				\end{itemize}    				
			\end{block}
		\end{column}
	\end{columns}
	\vfill
	\centering
	$\Rightarrow$ \alert{GTOs usually win but we use more!}
\end{frame}

\begin{frame}{The contraction problem}
	\begin{columns}
		\begin{column}{0.7\textwidth}
			\begin{block}{Primitive vs Contracted}
				\begin{itemize}
					\item Same center $\bA$
					\item Same angular momentum $\ba$
					\item Different exponent $\violet{\alpha_k}$
					\item Contraction coefficient $\blue{D_k}$ and degree $K$
				\end{itemize}
				\begin{equation}
					\underbrace{\braket{\red{\ba}\blue{\bb}}{\orange{\bc}\green{\bd}}}_{\text{\green{contracted ERI}}} 
					= \sum_{k_1}^{K_1} \sum_{k_2}^{K_2} \sum_{k_3}^{K_3} \sum_{k_4}^{K_4} 
					\blue{D_{k_1} D_{k_2} D_{k_3} D_{k_4}}
					\underbrace{\sbraket{\red{\ba}_{k_1}\blue{\bb}_{k_2}}{\orange{\bc}_{k_3}\green{\bd}_{k_4}}}_{\text{\red{primitive ERI}}} 
				\end{equation}
				\centering
				\green{One} contracted ERI required \red{$K_1 \times K_2 \times K_3 \times K_4$} primitive ERIs!
			\end{block}
			\begin{block}{Dunning's cc-pVTZ basis for the carbon atom}
				\begin{equation}
					\green{\braket{1s1s}{1s1s}}

\begin{frame}{Choose the basis for the physics you need}
	\begin{block}{Basis set choice is tricky}
    	\begin{itemize}
 			\bigskip
   		\item \textbf{Valence flexibility:} use double-, triple-, or higher-$\zeta$ descriptions to give each valence orbital several radial shapes
			\bigskip
    		\item \textbf{Polarization functions:} add \alert{higher angular momenta} to describe anisotropic density, bonding, and angular correlation
			\bigskip
    		\item \textbf{Diffuse functions:} add \alert{small exponents} to describe long-range density
    		\\
    		(anions, Rydberg states, noncovalent interactions, multipoles, (hyper)polarizabilities, etc)
			\bigskip
    		\item \textbf{Tight functions:} add \alert{large exponents} to descrive core region
    		\\
    		(core correlation, nuclear magnetic shielding, spin-spin coupling constants, etc)
			\bigskip
    	\end{itemize}
	\end{block}
	\vfill
	\centering
	$\Rightarrow$ \alert{The appropriate basis depends on both the electronic structure method and the property of interest!}
\end{frame}

\begin{frame}{Properties of Gaussian functions}
	\begin{block}{Gaussian product rule: \textit{``The product of two gaussians is a gaussian''}}
		\begin{equation}
			G_{\red{\alpha},\red{\bA}}(\br) = \exp(-\red{\alpha} \abs{\br - \red{\bA}}^2)	
			\qqtext{and}
			G_{\blue{\beta},\blue{\bB}}(\br) = \exp(-\blue{\beta} \abs{\br - \blue{\bB}}^2)	
			\qqtext{then}
		\end{equation}	
		\begin{equation}
		\boxed{G_{\red{\alpha},\red{\bA}}(\br) G_{\blue{\beta},\blue{\bB}}(\br) = K_{\violet{\zeta},\violet{\bP}} G_{\violet{\zeta},\violet{\bP}}(\br)}
		\qqtext{with} 
		\violet{\zeta} = \red{\alpha} + \blue{\beta} 
		\qqtext{and} 
		\violet{\bP} = \frac{\red{\alpha \bA} + \blue{\beta \bB}}{\red{\alpha} + \blue{\beta} } 
		\end{equation}		
		\begin{equation}
			K_{\violet{\zeta},\violet{\bP}} = \exp( -\frac{\red{\alpha} \blue{\beta}}{\red{\alpha} + \blue{\beta} } \abs{\red{\bA} - \blue{\bB}}^2)
			\qq{$\Rightarrow$}
			\text{decay exponentially fast with $\abs{\red{\bA} - \blue{\bB}}$}
		\end{equation}		
	\end{block}
	\begin{block}{Overalp}
		\begin{equation}
			\begin{split}
			\rbraket{\red{\ba}}{ \blue{\bb} }
			& = \iint G_{\red{\alpha},\red{\bA}}(\br) G_{\blue{\beta},\blue{\bB}}(\br) \dd{\br}
			= K_{\violet{\zeta},\violet{\bP}} \int G_{\violet{\zeta},\violet{\bP}}(\br) \dd{\br}
			\\
			& = K_{\violet{\zeta},\violet{\bP}} 
			\int_{-\infty}^{+\infty} e^{-\violet{\zeta} x^2} \dd{x} 
			\int_{-\infty}^{+\infty} e^{-\violet{\zeta} y^2} \dd{y} 
			\int_{-\infty}^{+\infty} e^{-\violet{\zeta} z^2} \dd{z} 
			\\
			& = K_{\violet{\zeta},\violet{\bP}} 
			\qty( \frac{\pi}{\violet{\zeta}})^{3/2}
			\qq{$\Rightarrow$}
			\text{\alert{cartesian factorization!}}
		\end{split}	
		\end{equation}	
	\end{block}
\end{frame}

\begin{frame}{Four-center electron repulsion integrals}
	\begin{block}{Gaussian product rule for ERIs}
		\alert{Formally, one has to compute $\order{N^4}$ ERIs!}
		\begin{equation}
			\begin{split}
			\rbraket{\red{\ba} \blue{\bb} }{ \orange{\bc} \green{\bd} }
			& = \iint G_{\red{\alpha},\red{\bA}}(\br_1) G_{\blue{\beta},\blue{\bB}}(\br_1) \frac{1}{r_{12}} G_{\orange{\gamma},\orange{\bC}}(\br_2) G_{\green{\delta},\green{\bD}}(\br_2) \dd{\br_1} \dd{\br_2} 
			\\
			& = K_{\violet{\zeta},\violet{\bP}} K_{\purple{\eta},\purple{\bQ}} \iint G_{\violet{\zeta},\violet{\bP}}(\br_1) \frac{1}{r_{12}} G_{\purple{\eta},\purple{\bQ}}(\br_2) \dd{\br_1} \dd{\br_2} 
		\end{split}	
		\end{equation}	
		\alert{The number of ``significant'' ERIs in a large system is $\order{N^2}$!}
	\end{block}
\end{frame}

    			& = K_{\violet{\zeta},\violet{\bP}} K_{\purple{\eta},\purple{\bQ}} \iint G_{\violet{\zeta},\violet{\bP}}(\br_1) \frac{1}{r_{12}} G_{\purple{\eta},\purple{\bQ}}(\br_2) \dd{\br_1} \dd{\br_2}

			= \frac{2}{\sqrt{\pi}} \int_0^\infty e^{- t^2 (x_1 - x_2)^2} e^{- t^2 (y_1 - y_2)^2} e^{- t^2 (z_1 - z_2)^2} \dd{t}
    	\end{equation}
	\end{block}
	    	\begin{equation}

\begin{frame}{Boys function}
	\begin{block}{Generalized Boys function}
		\begin{equation}
			F_m(x) = \int_0^1 t^{2m} \exp(-x t^2) \dd{x}
			\qquad
			F_0(x) = \sqrt{\frac{\pi}{4x}} \erf(\sqrt{x})
		\end{equation}
	\end{block}
	\begin{block}{Computation}
		\begin{itemize}
			\item $F_m(x)$ computed using pretabulated values on a grid and Taylor expansion (Chebyshev interpolation table)
			\item higher $m$ values computed via downward recursion
		\end{itemize}
	\end{block}
	\begin{block}{Downward recursion}
		\begin{equation}
			F_m(x) = \frac{2x F_{m+1} (x) + \exp(-x)}{2m+1}
			\qquad
			F_\infty(x) = 0
		\end{equation}
	\end{block}
	\begin{block}{Asymptotic behavior}
		\begin{equation}
			F_m(x) \sim \frac{(2m-1)!!}{2^{m+1}} \sqrt{\frac{\pi}{x^{2m+1}}}
			\qq{for large $x$}
		\end{equation}
	\end{block}
\end{frame}

\begin{frame}{Cauchy-Schwartz bound}

		= \abs{\rbraket{\violet{\bP}}{\purple{\bQ}}}
		\le

		\sqrt{\rbraket{\purple{\bQ}}{\purple{\bQ}}}
	\end{equation}
	\begin{itemize}
		\item $\rbraket{\violet{\bP}}{\purple{\bQ}}$ should decay as $1/\abs{\violet{\bP} - \purple{\bQ}}$ or faster
		\item However, there is no distance information in the product $\rbraket{\violet{\bP}}{\violet{\bP}} \rbraket{\purple{\bQ}}{\purple{\bQ}}$
		\item The Cauchy-Schwarz bound is rigorous and cheap but is not tight for well-separated shell pairs
	\end{itemize}
\end{frame}


\begin{frame}{A tale of three strategies}
\begin{itemize}
	\item \orange{\bf Integral bound}
		\\
		We seek a single number that bounds a particular integral
		\begin{equation*}
			\abs{\sbraket{\ba_1\ba_2}{\bb_1\bb_2}} \le \sbraket{\ba_1\ba_1}{\bb_1\bb_1}^{1/2} \sbraket{\ba_2\ba_2}{\bb_2\bb_2}^{1/2}
		\end{equation*}
		With $\order{N^{2}}$ $\sbraket{\ba_1\ba_1}{\bb_1\bb_1}$ bound factors, we can upper bound $\order{N^{4}}$ $\sbraket{\ba_1\ba_2}{\bb_1\bb_2}$ integrals

	\item \alert{\bf Non-factorizable class bound}
		\\
		We seek a single number that bounds an entire class*
		\begin{equation*}
			\abs{\sbraket{a_1a_2}{b_1b_2}} \le \sbraket{\sba_{1} \sba_{2}}{\sbb_{1} \sbb_{2}}
		\end{equation*}
	\item \violet{\bf Factorizable class bound}
		\\
		We combine the two previous ideas
		\begin{equation*}
			\begin{split}
				\left| \sbraket{a_1a_2}{b_1b_2}  \right| 
				& \le \sbraket{\sba_{1} \sba_{2}}{\sbb_{1} \sbb_{2}}
				\\
				& \le \sbraket{\sba_1\sba_1}{\sbb_1\sbb_1}^{1/2} \sbraket{\sba_2\sba_2}{\sbb_2\sbb_2}^{1/2}
			\end{split}
		\end{equation*}
\end{itemize}
*{\small The simple $\sbraket{dd}{dd}$ class is made of 625 integrals!}
\end{frame}

\begin{frame}{Computation of the Coulomb matrix}
	\begin{equation}
		\boxed{
		J_{\red{\mu}\blue{\nu}} = \sum_{\red{\mu}\blue{\nu}} P_{\orange{\la}\violet{\si}} \rbraket{\red{\mu}\blue{\nu} }{ \orange{\la}\violet{\si}}
		} 
	\end{equation}
	\begin{algorithmic}
		\Procedure{Computing the Coulomb matrix}{}
			\For{$\red{\mu}=1,N$}
				\For{$\blue{\nu}=1,N$}
					\State $J_{\red{\mu}\blue{\nu}} = 0$ \Comment{Initialization of the array}
					\For{$\orange{\la}=1,N$}
						\For{$\violet{\si}=1,N$}
							\State $J_{\red{\mu}\blue{\nu}} 
							= J_{\red{\mu}\blue{\nu}} 
							+ P_{\orange{\la}\violet{\si}} \rbraket{\red{\mu}\blue{\nu} }{ \orange{\la}\violet{\si} }$ 
							\Comment{Accumulation step}
						\EndFor
					\EndFor 
				\EndFor
			\EndFor 
		\EndProcedure
		\Comment{\red{This is a $\order{N^4}$ algorithm as it involves four loops}}
	\end{algorithmic}
\end{frame}

\begin{frame}{Resolution of the identity}
	\begin{block}{Resolution of the identity (RI)}
		\begin{equation}
			\sum_{\green{A}=1}^{\red{\infty}} \dyad{\green{A}} = \hI
			\qq{with}
			\braket{\green{A}}{\green{B}} = \delta_{AB}
			\qq{$\Leftrightarrow$}
			\sum_{\green{A}=1}^{\red{\infty}} \green{A}(\br_1) \green{A}(\br_2)  
			= \delta(\br_1 - \br_2) 
		\end{equation}
	\end{block}
	\begin{block}{Generalization to a two-body operator $\hO$}
		\begin{equation}
			\sum_{\green{\Tilde{A}}=1}^{\red{\infty}} \dyad{\green{\Tilde{A}}} = \hO
			\qq{with}
			\mel{\green{A}}{\hO}{\green{B}} = \delta_{AB}
			\qq{and}
			\hO \ket{\green{A}} = \ket{\green{\Tilde{A}}}
			\qq{$\Leftrightarrow$}
			\sum_{\green{\Tilde{A}}=1}^{\red{\infty}} \green{\Tilde{A}}(\br_1) \green{\Tilde{A}}(\br_2)  
			= \hO(\br_1,\br_2) 
		\end{equation}
	\end{block}

\begin{frame}{Resolution of the Coulomb operator}
	\begin{block}{RI in practice = RI \alert{approximation}}
		\begin{equation}
			\boxed{\sum_{\green{A}=1}^{\red{\infty}} \dyad{\green{A}} = \hI

			\sum_{\green{A}=1}^{\red{K}} \dyad{\green{A}} \approx \hI}

	\begin{block}{Computing the Coulomb matrix within the RI approximation}
		\begin{equation}
			\begin{split}
				J_{\red{\mu\nu}} 
				& = \sum_{\blue{\la \si}} P_{\blue{\la\si}} (\red{\mu\nu}|\blue{\la\si})
				\\
				& \stackrel{\text{\green{RI}}}{=}  \sum_{\blue{\la \si}} P_{\blue{\la\si}} \sum_{\green{A}} (\red{\mu\nu}|\green{A}) (\green{A}|\blue{\la\si})
				\\
				& = \sum_{\green{A}} (\red{\mu\nu}|\green{A})  
				\underbrace{\sum_{\blue{\la \si}} P_{\blue{\la\si}} (\green{A}|\blue{\la\si})}_{\order{KN^2} \text{ and $K$ storage}}
				= \underbrace{\sum_{\green{A}} (\red{\mu\nu}|\green{A})  \rho_{\green{A}}}_{\order{KN^2}}
			\end{split}
